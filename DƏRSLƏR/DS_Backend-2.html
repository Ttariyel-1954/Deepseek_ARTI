1 Ortaq səhifələmə DTO-su

A · Bu addım nəyə görədir

Hər siyahı endpoint-i eyni suallara cavab verməlidir: cəmi neçə sətir, hansı səhifə, neçə səhifə. Bu cavabları hər modulda ayrı-ayrı yazsaq, frontend hər endpoint üçün fərqli kod yazmalı olar. Ona görə səhifələməni bir dəfə ortaq DTO-da yazırıq və bütün filtr DTO-ları ondan miras alır. Bu addımda həm sorğu parametrləri (SehifeSorghu), həm cavab forması (Sehifelenmis<T>), həm də iki köməkçi funksiya yaranır.

import { ApiPropertyOptional } from '@nestjs/swagger';
import { Type } from 'class-transformer';
import { IsIn, IsInt, IsOptional, IsString, Max, MaxLength, Min } from 'class-validator';

/**
 * Səhifələnmiş cavab — BÜTÜN siyahı endpoint-ləri bu formanı qaytarır.
 * Frontend bir dənə cədvəl komponenti ilə hər siyahını göstərə bilir.
 */
export interface Sehifelenmis<T> {
  setirler: T[];
  cemi: number;
  sehife: number;
  limit: number;
  sehife_sayi: number;
}

/**
 * Səhifələmə, axtarış və sıralama üçün ORTAQ parametrlər.
 * Bütün filtr DTO-ları bundan miras alır — təkrar yazmaq lazım deyil.
 */
export class SehifeSorghu {
  @ApiPropertyOptional({ default: 1, minimum: 1 })
  @IsOptional()
  @Type(() => Number)
  @IsInt({ message: 'sehife tam ədəd olmalıdır' })
  @Min(1, { message: 'sehife ən azı 1 olmalıdır' })
  sehife: number = 1;

  @ApiPropertyOptional({ default: 20, minimum: 1, maximum: 100 })
  @IsOptional()
  @Type(() => Number)
  @IsInt({ message: 'limit tam ədəd olmalıdır' })
  @Min(1, { message: 'limit ən azı 1 olmalıdır' })
  @Max(100, { message: 'limit 100-dən çox ola bilməz' })
  limit: number = 20;

  @ApiPropertyOptional({ description: 'Axtarış mətni' })
  @IsOptional()
  @IsString()
  @MaxLength(80, { message: 'axtar ən çoxu 80 simvol ola bilər' })
  axtar?: string;

  @ApiPropertyOptional({ enum: ['asc', 'desc'], default: 'asc' })
  @IsOptional()
  @IsIn(['asc', 'desc'], { message: "siralama yalnız 'asc' və ya 'desc' ola bilər" })
  siralama: 'asc' | 'desc' = 'asc';
}

/** Səhifəni təhlükəsiz hədlərə salır və SQL üçün hazırlayır */
export function sehifeHesabla(sehife = 1, limit = 20): {
  sehife: number; limit: number; skip: number;
} {
  const s = Math.max(1, Math.trunc(Number(sehife) || 1));
  const l = Math.min(100, Math.max(1, Math.trunc(Number(limit) || 20)));
  return { sehife: s, limit: l, skip: (s - 1) * l };
}

/** Səhifələnmiş cavabı yığır */
export function sehifelenmis<T>(
  setirler: T[],
  cemi: number,
  sehife: number,
  limit: number,
): Sehifelenmis<T> {
  return {
    setirler,
    cemi,
    sehife,
    limit,
    // 0 sətir olsa da ən azı 1 səhifə var
    sehife_sayi: Math.max(1, Math.ceil(cemi / limit)),
  };
}

2 Struktur filtri DTO-su

A · Bu addım nəyə görədir

Mərkəzlər siyahısında istifadəçi tipə, aktivliyə görə süzgəc qoymalı və istədiyi sütun üzrə sıralamalıdır. Bu sahələrin adları birbaşa SQL/Prisma sorğusuna düşdüyü üçün onlar mütləq yoxlanılmalıdır — əks halda istifadəçi ?tip=<script> kimi dəyər göndərə bilər. Bu DTO icazə verilən tipləri və sıralama sütunlarını sabit siyahı kimi elan edir və @IsIn ilə onları yoxlayır.

import { ApiPropertyOptional } from '@nestjs/swagger';
import { IsBooleanString, IsIn, IsOptional } from 'class-validator';
import { SehifeSorghu } from '../../common/dto/sehife.dto.js';

export const MERKEZ_TIPLERI = ['merkez', 'katiblik', 'sobe', 'sektor'] as const;
export type MerkezTipi = (typeof MERKEZ_TIPLERI)[number];

export const MERKEZ_SAHELERI = ['ad', 'tip', 'id', 'yaradilma_tarixi'] as const;

/**
 * Mərkəzlər siyahısı üçün filtr.
 * `sirala` — hansı sütun üzrə, `siralama` — hansı istiqamətdə (asc/desc).

export class MerkezFiltrDto extends SehifeSorghu {
  @ApiPropertyOptional({ enum: MERKEZ_TIPLERI, description: 'Mərkəz tipi' })
  @IsOptional()
  @IsIn(MERKEZ_TIPLERI as unknown as string[], {
    message: `tip yalnız bunlardan biri ola bilər: ${MERKEZ_TIPLERI.join(', ')}`,
  })
  tip?: MerkezTipi;

  @ApiPropertyOptional({ enum: ['true', 'false'], description: 'Aktiv/passiv' })
  @IsOptional()
  @IsBooleanString({ message: "aktiv yalnız 'true' və ya 'false' ola bilər" })
  aktiv?: string;

  @ApiPropertyOptional({ enum: MERKEZ_SAHELERI, default: 'ad' })
  @IsOptional()
  @IsIn(MERKEZ_SAHELERI as unknown as string[], {
    message: `sirala yalnız bunlardan biri ola bilər: ${MERKEZ_SAHELERI.join(', ')}`,
  })
  sirala?: MerkezSahesi;
}

import { ConflictException, Injectable, NotFoundException } from '@nestjs/common';
import { Prisma } from '../generated/prisma/client.js';
import { PrismaService } from '../prisma/prisma.service.js';
import {
  Sehifelenmis, sehifeHesabla, sehifelenmis,
} from '../common/dto/sehife.dto.js';
import { MerkezFiltrDto } from './dto/merkez-filtr.dto.js';
import { CreateMerkezDto } from './dto/create-merkez.dto.js';
import { UpdateMerkezDto } from './dto/update-merkez.dto.js';

export interface Merkez {
  id: number;
  ad: string;
  tip: string;
  tesvir: string | null;
  unvan: string | null;
  telefon: string | null;
  email: string | null;
  yaradilma_tarixi: Date | null;
  aktiv: boolean;
}

export interface MerkezStatistikasi {
  merkez_id: number;
  ad: string;
  shobe_sayi: number;
  emekdas_sayi: number;
}

export interface SilmeNetici {
  silindi: number;
  ad: string;
}

@Injectable()
export class StrukturService {
  constructor(private readonly prisma: PrismaService) {}

  /** Mərkəzlərin səhifələnmiş siyahısı — filtr, axtarış və sıralama ilə */
  async merkezler(filtr: MerkezFiltrDto): Promise<Sehifelenmis<Merkez>> {
    const { sehife, limit, skip } = sehifeHesabla(filtr.sehife, filtr.limit);

    const where: Prisma.merkezlerWhereInput = {
      ...(filtr.tip ? { tip: filtr.tip } : {}),
      ...(filtr.aktiv !== undefined ? { aktiv: filtr.aktiv === 'true' } : {}),
      ...(filtr.axtar
        ? { ad: { contains: filtr.axtar, mode: 'insensitive' as const } }
        : {}),
    };

    // Sütun adı istifadəçidən gəlir — DTO onu @IsIn ilə YOXLAYIR,
    // ona görə burada SQL inyeksiyası mümkün deyil.
    const orderBy = {
      [filtr.sirala ?? 'ad']: filtr.siralama,
    } as Prisma.merkezlerOrderByWithRelationInput;

    const [setirler, cemi] = await Promise.all([
      this.prisma.merkezler.findMany({ where, orderBy, skip, take: limit }),
      this.prisma.merkezler.count({ where }),
    ]);

    return sehifelenmis(setirler as Merkez[], cemi, sehife, limit);
  }

  /** Mərkəz üzrə şöbə və əməkdaş sayı */
  async merkezStatistikasi(): Promise<MerkezStatistikasi[]> {
    // count() PostgreSQL-də bigint qaytarır — ::int olmasa
    // JSON-a çevriləndə "Do not know how to serialize a BigInt" xətası çıxır.
    return this.prisma.$queryRaw<MerkezStatistikasi[]>`
      SELECT m.id                     AS merkez_id,
             m.ad,
             count(DISTINCT s.id)::int AS shobe_sayi,
             count(DISTINCT e.id)::int AS emekdas_sayi
        FROM struktur.merkezler m
        LEFT JOIN struktur.shobeler   s ON s.merkez_id = m.id AND s.aktiv
        LEFT JOIN kadrlar.emekdaslar  e ON e.merkez_id = m.id AND e.aktiv
       GROUP BY m.id, m.ad
       ORDER BY m.ad`;
  }

  /** Bir mərkəz — tapılmasa 404 */
  async merkez(id: number): Promise<Merkez> {
    const setir = await this.prisma.merkezler.findUnique({ where: { id } });

    if (!setir) {
      throw new NotFoundException(`${id} nömrəli mərkəz tapılmadı`);
    }
    return setir as Merkez;
  }

  /** Yeni mərkəz — ad təkrarı 409 verir */
  async yarat(dto: CreateMerkezDto): Promise<Merkez> {
    try {
      return (await this.prisma.merkezler.create({
        data: {
          ad: dto.ad,
          tip: dto.tip ?? 'merkez',
          tesvir: dto.tesvir ?? null,
          unvan: dto.unvan ?? null,
          telefon: dto.telefon ?? null,
          email: dto.email ?? null,
          yaradilma_tarixi: dto.yaradilma_tarixi
            ? new Date(dto.yaradilma_tarixi)
            : null,
          aktiv: dto.aktiv ?? true,
        },
      })) as Merkez;
    } catch (xeta) {
      if (
        xeta instanceof Prisma.PrismaClientKnownRequestError &&
        xeta.code === 'P2002'
      ) {
        throw new ConflictException(`«${dto.ad}» adlı mərkəz artıq mövcuddur`);
      }
      throw xeta;
    }
  }

  /** Mərkəzi yenilə — yalnız göndərilən sahələr dəyişir */
  async yenile(id: number, dto: UpdateMerkezDto): Promise<Merkez> {
    await this.merkez(id); // yoxdursa 404

    try {
      return (await this.prisma.merkezler.update({
        where: { id },
        data: {
          ...(dto.ad !== undefined ? { ad: dto.ad } : {}),
          ...(dto.tip !== undefined ? { tip: dto.tip } : {}),
          ...(dto.tesvir !== undefined ? { tesvir: dto.tesvir } : {}),
          ...(dto.unvan !== undefined ? { unvan: dto.unvan } : {}),
          ...(dto.telefon !== undefined ? { telefon: dto.telefon } : {}),
          ...(dto.email !== undefined ? { email: dto.email } : {}),
          ...(dto.yaradilma_tarixi !== undefined
            ? { yaradilma_tarixi: new Date(dto.yaradilma_tarixi) }
            : {}),
          ...(dto.aktiv !== undefined ? { aktiv: dto.aktiv } : {}),
        },
      })) as Merkez;
    } catch (xeta) {
      if (
        xeta instanceof Prisma.PrismaClientKnownRequestError &&
        xeta.code === 'P2002'
      ) {
        throw new ConflictException(`«${dto.ad}» adlı mərkəz artıq mövcuddur`);
      }
      throw xeta;
    }
  }

  /**
   * Mərkəzi sil.
   *
   * ƏVVƏLCƏ bağlı sətirlər yoxlanılır. Səbəb: `shobeler` cədvəlində
   * ON DELETE CASCADE var, amma `shobeler`-ə bağlı `emekdaslar` YOXDUR.
   * Yəni kaskad işə düşsə, əməkdaşların şöbəsi silinərkən xəta çıxar və
   * istifadəçi anlaşılmaz 500 görər. Biz əvvəlcədən yoxlayıb aydın 409 veririk.
   */
  async sil(id: number): Promise<SilmeNetici> {
    const merkez = await this.merkez(id);

    const [shobe, kadr, rehber] = await Promise.all([
      this.prisma.shobeler.count({ where: { merkez_id: id } }),
      this.prisma.emekdaslar.count({ where: { merkez_id: id } }),
      this.prisma.rehberlik.count({ where: { merkez_id: id } }),
    ]);

    const mane = [
      shobe > 0 ? `${shobe} şöbə` : null,
      kadr > 0 ? `${kadr} əməkdaş` : null,
      rehber > 0 ? `${rehber} rəhbərlik qeydi` : null,
    ].filter((s): s is string => s !== null);

    if (mane.length > 0) {
      throw new ConflictException(
        `«${merkez.ad}» silinmir — ona bağlı ${mane.join(', ')} var`,
      );
    }

    await this.prisma.merkezler.delete({ where: { id } });
    return { silindi: id, ad: merkez.ad };
  }
}

import { Injectable, NotFoundException } from '@nestjs/common';
import { Prisma } from '../generated/prisma/client.js';
import { PrismaService } from '../prisma/prisma.service.js';
import {
  Sehifelenmis, sehifeHesabla, sehifelenmis,
} from '../common/dto/sehife.dto.js';
import { EmekdasFiltrDto } from './dto/emekdas-filtr.dto.js';

/** Əməkdaşın tam profili — bütün axtarış cədvəlləri açılmış şəkildə */
export interface EmekdasTam {
  id: number;
  ad: string;
  soyad: string;
  ata_adi: string;
  email: string | null;
  telefon: string | null;
  maas: number | null;
  ise_baslama: string | null;
  aktiv: boolean;
  merkez: string | null;
  shobe: string | null;
  vezife: string | null;
  elmi_derece: string | null;
  elmi_ad: string | null;
  is_statusu: string | null;
}

export interface IcmalSetri {
  ad: string;
  say: number;
}

export interface KadrIcmali {
  cemi: number;
  aktiv: number;
  merkez_uzre: IcmalSetri[];
  vezife_uzre: IcmalSetri[];
}

@Injectable()
export class KadrlarService {
  constructor(private readonly prisma: PrismaService) {}

  /**
   * Filtr şərtini BİR DƏFƏ qurub həm siyahıda, həm saymada işlədirik.
   * Əks halda iki sorğunun şərti zamanla bir-birindən ayrı düşər.
   */
  private serh(f: EmekdasFiltrDto): Prisma.Sql {
    const axtar = f.axtar ? `%${f.axtar}%` : null;
    const merkezId = f.merkez_id ?? null;
    const shobeId = f.shobe_id ?? null;
    const aktiv = f.aktiv === undefined ? null : f.aktiv === 'true';

    return Prisma.sql`
      (${axtar}::text IS NULL
        OR e.ad ILIKE ${axtar}::text
        OR e.soyad ILIKE ${axtar}::text
        OR e.ata_adi ILIKE ${axtar}::text)
      AND (${merkezId}::int IS NULL OR e.merkez_id = ${merkezId}::int)
      AND (${shobeId}::int IS NULL OR e.shobe_id = ${shobeId}::int)
      AND (${aktiv}::boolean IS NULL OR e.aktiv = ${aktiv}::boolean)`;
  }

  /** Əməkdaşların səhifələnmiş siyahısı — tam profillə */
  async emekdaslar(f: EmekdasFiltrDto): Promise<Sehifelenmis<EmekdasTam>> {
    const { sehife, limit, skip } = sehifeHesabla(f.sehife, f.limit);
    const serh = this.serh(f);

    const setirler = await this.prisma.$queryRaw<EmekdasTam[]>`
      SELECT e.id::int              AS id,
             e.ad, e.soyad, e.ata_adi, e.email, e.telefon,
             e.maas::float8         AS maas,
             e.ise_baslama::text    AS ise_baslama,
             e.aktiv,
             m.ad                   AS merkez,
             s.ad                   AS shobe,
             v.ad                   AS vezife,
             d.ad                   AS elmi_derece,
             a.ad                   AS elmi_ad,
             st.ad                  AS is_statusu
        FROM kadrlar.emekdaslar e
        LEFT JOIN struktur.merkezler    m  ON m.id = e.merkez_id
        LEFT JOIN struktur.shobeler     s  ON s.id = e.shobe_id
        LEFT JOIN struktur.vezifeler    v  ON v.id = e.vezife_id
        LEFT JOIN ortaq.elmi_dereceler  d  ON d.id = e.elmi_derece_id
        LEFT JOIN ortaq.elmi_adlar      a  ON a.id = e.elmi_ad_id
        LEFT JOIN ortaq.is_statuslari   st ON st.id = e.is_status_id
       WHERE ${serh}
       ORDER BY e.soyad ${f.siralama === 'desc' ? Prisma.sql`DESC` : Prisma.sql`ASC`},
                e.ad ASC
       LIMIT ${limit} OFFSET ${skip}`;

    const cem = await this.prisma.$queryRaw<{ cemi: number }[]>`
      SELECT count(*)::int AS cemi
        FROM kadrlar.emekdaslar e
       WHERE ${serh}`;

    return sehifelenmis(setirler, cem[0]?.cemi ?? 0, sehife, limit);
  }

  /** Bir əməkdaşın tam profili */
  async emekdas(id: number): Promise<EmekdasTam> {
    const setirler = await this.prisma.$queryRaw<EmekdasTam[]>`
      SELECT e.id::int              AS id,
             e.ad, e.soyad, e.ata_adi, e.email, e.telefon,
             e.maas::float8         AS maas,
             e.ise_baslama::text    AS ise_baslama,
             e.aktiv,
             m.ad                   AS merkez,
             s.ad                   AS shobe,
             v.ad                   AS vezife,
             d.ad                   AS elmi_derece,
             a.ad                   AS elmi_ad,
             st.ad                  AS is_statusu
        FROM kadrlar.emekdaslar e
        LEFT JOIN struktur.merkezler    m  ON m.id = e.merkez_id
        LEFT JOIN struktur.shobeler     s  ON s.id = e.shobe_id
        LEFT JOIN struktur.vezifeler    v  ON v.id = e.vezife_id
        LEFT JOIN ortaq.elmi_dereceler  d  ON d.id = e.elmi_derece_id
        LEFT JOIN ortaq.elmi_adlar      a  ON a.id = e.elmi_ad_id
        LEFT JOIN ortaq.is_statuslari   st ON st.id = e.is_status_id
       WHERE e.id = ${id}
       LIMIT 1`;

    const setir = setirler[0];
    if (!setir) {
      throw new NotFoundException(`${id} nömrəli əməkdaş tapılmadı`);
    }
    return setir;
  }

  /** Kadr icmalı — mərkəz və vəzifə üzrə bölgü */
  async icmal(): Promise<KadrIcmali> {
    const [cem, merkezUzre, vezifeUzre] = await Promise.all([
      this.prisma.$queryRaw<{ cemi: number; aktiv: number }[]>`
        SELECT count(*)::int                        AS cemi,
               count(*) FILTER (WHERE e.aktiv)::int AS aktiv
          FROM kadrlar.emekdaslar e`,

      this.prisma.$queryRaw<IcmalSetri[]>`
        SELECT COALESCE(m.ad, '— təyin edilməyib —') AS ad,
               count(*)::int                          AS say
          FROM kadrlar.emekdaslar e
          LEFT JOIN struktur.merkezler m ON m.id = e.merkez_id
         GROUP BY m.ad
         ORDER BY count(*) DESC, m.ad`,

      this.prisma.$queryRaw<IcmalSetri[]>`
        SELECT v.ad,
               count(*)::int AS say
          FROM kadrlar.emekdaslar e
          JOIN struktur.vezifeler v ON v.id = e.vezife_id
         GROUP BY v.ad
         ORDER BY count(*) DESC, v.ad
         LIMIT 10`,
    ]);

    return {
      cemi: cem[0]?.cemi ?? 0,
      aktiv: cem[0]?.aktiv ?? 0,
      merkez_uzre: merkezUzre,
      vezife_uzre: vezifeUzre,
    };
  }
}
