import { ApiProperty, ApiPropertyOptional } from '@nestjs/swagger';
import {
  IsEmail, IsIn, IsInt, IsOptional, IsString, MaxLength, MinLength,
} from 'class-validator';

/** Sistemdə mövcud olan DÖRD rol */
export const ROLLAR = ['admin', 'muhendis', 'maliyyeci', 'baxici'] as const;
export type Rol = (typeof ROLLAR)[number];

export class QeydiyyatDto {
  @ApiProperty({ example: 'yeni@arti.edu.az' })
  @IsEmail({}, { message: 'E-poçt ünvanı yanlışdır' })
  @MaxLength(120)
  email!: string;

  @ApiProperty({ example: 'guclu-sifre-2026', minLength: 8 })
  @IsString()
  @MinLength(8, { message: 'Şifrə ən azı 8 simvol olmalıdır' })
  @MaxLength(100)
  parol!: string;

  @ApiProperty({ example: 'Əliyev Kamran' })
  @IsString()
  @MinLength(3, { message: 'Ad-soyad ən azı 3 simvol olmalıdır' })
  @MaxLength(150)
  ad_soyad!: string;

  @ApiPropertyOptional({ enum: ROLLAR, default: 'baxici' })
  @IsOptional()
  @IsIn(ROLLAR as unknown as string[], {
    message: `rol yalnız bunlardan biri ola bilər: ${ROLLAR.join(', ')}`,
  })
  rol?: Rol;

  @ApiPropertyOptional({ description: 'Bağlı əməkdaş id (varsa)' })
  @IsOptional()
  @IsInt({ message: 'emekdas_id tam ədəd olmalıdır' })
  emekdas_id?: number;
}
EOF

3 Üç dekorator — @Public, @Roles, @CurrentUser

A · Bu addım nəyə görədir

Bu üç dekorator guard-larla controller arasında əlaqə dilidir. @Public() "bu endpoint-i qoruma" deyir, @Roles('admin') "yalnız bu rollar girə bilər", @CurrentUser() isə JWT-dən çıxarılan istifadəçini metodun parametrinə ötürür. Onlar sadəcə SetMetadata yazır — qərarı guard-lar verir. Bu ayrılıq vacibdir: controller heç vaxt icazə yoxlamır.

import { createParamDecorator, ExecutionContext } from '@nestjs/common';

/** JWT-dən çıxarılan istifadəçi məlumatı */
export interface CariIstifadeci {
  id: number;
  email: string;
  ad_soyad: string;
  rol: string;
}

/**
 * Cari istifadəçini controller metoduna ötürür.
 *
 *   @Get('profil')
 *   profil(@CurrentUser() istifadeci: CariIstifadeci) { ... }
 */
export const CurrentUser = createParamDecorator(
  (sahe: keyof CariIstifadeci | undefined, ctx: ExecutionContext) => {
    const sorqu = ctx.switchToHttp().getRequest();
    const istifadeci = sorqu.user as CariIstifadeci | undefined;
    if (!istifadeci) return undefined;
    return sahe ? istifadeci[sahe] : istifadeci;
  },
);
EOF

4 JwtStrategy — tokeni oxuyan strategiya

import { ExecutionContext, Injectable } from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { AuthGuard } from '@nestjs/passport';
import { PUBLIC_ACARI } from '../decorators/public.decorator.js';

/**
 * QLOBAL autentifikasiya guard-ı.
 *
 * Bütün endpoint-lər qorunur; @Public() ilə işarələnənlər istisnadır.
 * Bu, "təhlükəsiz susmaya görə" prinsipidir: yeni endpoint əlavə edəndə
 * onu qorumaq üçün heç nə etmək lazım deyil — susmaya görə qorunur.

@Injectable()
export class JwtAuthGuard extends AuthGuard('jwt') {
  constructor(private readonly reflector: Reflector) {
    super();
  }

  canActivate(context: ExecutionContext) {
    const publicdir = this.reflector.getAllAndOverride<boolean>(PUBLIC_ACARI, [
      context.getHandler(),
      context.getClass(),
    ]);

    if (publicdir) return true; // @Public() — burax
    return super.canActivate(context);
  }
}

import {
  CanActivate, ExecutionContext, ForbiddenException, Injectable,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { ROLLAR_ACARI } from '../decorators/roles.decorator.js';
import type { Rol } from '../dto/qeydiyyat.dto.js';

/**
 * RBAC — Role-Based Access Control.
 * @Roles(...) ilə işarələnmiş endpoint-ə yalnız həmin rollar girə bilər.
 */
@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private readonly reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const teleb = this.reflector.getAllAndOverride<Rol[]>(ROLLAR_ACARI, [
      context.getHandler(),
      context.getClass(),
    ]);

    // @Roles yoxdursa — hər autentifikasiya olunmuş istifadəçi girə bilər
    if (!teleb || teleb.length === 0) return true;

    const sorqu = context.switchToHttp().getRequest();
    const istifadeci = sorqu.user as { rol?: string } | undefined;

    if (!istifadeci?.rol) {
      throw new ForbiddenException('İstifadəçi rolu müəyyən deyil');
    }

    // 'admin' HƏR ŞEYƏ icazəlidir — super-rol
    if (istifadeci.rol === 'admin') return true;

    if (!teleb.includes(istifadeci.rol as Rol)) {
      throw new ForbiddenException(
        `Bu əməliyyat üçün icazəniz yoxdur. ` +
          `Tələb olunan rol: ${teleb.join(' və ya ')}. Sizin rol: ${istifadeci.rol}`,
      );
    }

    return true;
  }
}

6 AuthService — login, timing attack və qeydiyyat

A · Bu addım nəyə görədir

Servis qatı şifrə yoxlamasını və token imzalanmasını idarə edir. Ən incə hissə timing attack qorumasıdır: istifadəçi tapılmayanda da bcrypt işlədirik ki, cavab vaxtı hər iki halda eyni olsun. Bu olmasa hücumçu sadəcə vaxta baxaraq hansı e-poçtun sistemdə qeydiyyatda olduğunu öyrənə bilər. Servis həmçinin parol_hash-ı heç vaxt cavaba salmır.

import {
  ConflictException, Injectable, Logger, UnauthorizedException,
} from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { JwtService } from '@nestjs/jwt';
import * as bcrypt from 'bcryptjs';
import { Prisma } from '../generated/prisma/client.js';
import { PrismaService } from '../prisma/prisma.service.js';
import { LoginDto } from './dto/login.dto.js';
import { QeydiyyatDto } from './dto/qeydiyyat.dto.js';

/**
 * ⚠️ TIMING ATTACK QORUMASI
 *
 * İstifadəçi tapılmayanda da bcrypt işlətmək üçün saxta hash.
 * Bu olmasa "mövcud olmayan e-poçt" sorğusu 1 ms, "səhv şifrə" isə
 * ~100 ms çəkər — hücumçu vaxt fərqindən hansı e-poçtun mövcud
 * olduğunu təxmin edə bilər.
 */
const SAXTA_HASH = '$2b$10$/2prlrsWPGoln3sVls0Ffu11aVRZJQhpFQrjolEW0WW.wYvBfkWWS';

const BCRYPT_RAUND = 10;

const UMUMI_XETA = 'E-poçt və ya şifrə yanlışdır';

interface IstifadeciSetri {
  id: number;
  email: string;
  parol_hash: string;
  ad_soyad: string;
  rol: string;
  emekdas_id: number | null;
  aktiv: boolean;
}

export interface IstifadeciMelumati {
  id: number;
  email: string;
  ad_soyad: string;
  rol: string;
  emekdas_id: number | null;
}

export interface LoginCavabi {
  token: string;
  istifadeci: IstifadeciMelumati;
  bitme: string;
}

@Injectable()
export class AuthService {
  private readonly log = new Logger('AUTH');

  constructor(
    private readonly prisma: PrismaService,
    private readonly jwt: JwtService,
    private readonly config: ConfigService,
  ) {}

  async login(dto: LoginDto): Promise<LoginCavabi> {
    const setirler = await this.prisma.$queryRaw<IstifadeciSetri[]>`
      SELECT id::int          AS id,
             email,
             parol_hash,
             ad_soyad,
             rol,
             emekdas_id::int   AS emekdas_id,
             aktiv
        FROM kadrlar.istifadeciler
       WHERE email = lower(${dto.email})`;

    const istifadeci = setirler[0];

    // İstifadəçi tapılmasa da bcrypt işlədilir — vaxt fərqi olmasın
    const hash = istifadeci?.parol_hash ?? SAXTA_HASH;
    const duzdur = await bcrypt.compare(dto.parol, hash);

    if (!istifadeci || !duzdur) {
      this.log.warn(`Uğursuz giriş cəhdi: ${dto.email}`);
      throw new UnauthorizedException(UMUMI_XETA);
    }

    if (!istifadeci.aktiv) {
      throw new UnauthorizedException('İstifadəçi deaktiv edilib');
    }

    const token = await this.jwt.signAsync({
      sub: istifadeci.id,
      email: istifadeci.email,
      rol: istifadeci.rol,
    });

    this.log.log(`Giriş uğurlu: ${istifadeci.email} (${istifadeci.rol})`);

    return {
      token,
      istifadeci: {
        id: istifadeci.id,
        email: istifadeci.email,
        ad_soyad: istifadeci.ad_soyad,
        rol: istifadeci.rol,
        emekdas_id: istifadeci.emekdas_id,
      },
      bitme: this.config.get<string>('JWT_MUDDET') ?? '8h',
    };
  }

  /** Yeni istifadəçi — yalnız admin çağıra bilər (@Roles ilə qorunur) */
  async qeydiyyat(dto: QeydiyyatDto): Promise<IstifadeciMelumati> {
    const hash = await bcrypt.hash(dto.parol, BCRYPT_RAUND);

    try {
      const setirler = await this.prisma.$queryRaw<IstifadeciMelumati[]>`
        INSERT INTO kadrlar.istifadeciler
               (email, parol_hash, ad_soyad, rol, emekdas_id, aktiv)
        VALUES (lower(${dto.email}), ${hash}, ${dto.ad_soyad},
                ${dto.rol ?? 'baxici'}, ${dto.emekdas_id ?? null}, true)
        RETURNING id::int AS id, email, ad_soyad, rol,
                  emekdas_id::int AS emekdas_id`;

      this.log.log(`Yeni istifadəçi: ${dto.email} (${dto.rol ?? 'baxici'})`);
      return setirler[0];
    } catch (xeta) {
      if (
        xeta instanceof Prisma.PrismaClientKnownRequestError &&
        xeta.code === 'P2002'
      ) {
        throw new ConflictException(`«${dto.email}» artıq qeydiyyatdadır`);
      }
      throw xeta;
    }
  }

  /**
   * Bütün istifadəçilər — yalnız admin.
   * ⚠️ `parol_hash` siyahıya SALINMIR — heç vaxt.
   */
  async siyahi(): Promise<IstifadeciMelumati[]> {
    return this.prisma.$queryRaw<IstifadeciMelumati[]>`
      SELECT id::int        AS id,
             email,
             ad_soyad,
             rol,
             emekdas_id::int AS emekdas_id
        FROM kadrlar.istifadeciler
       ORDER BY id`;
  }
}

8 ⚠️ AuthModule — registerAsync, register() YOX

A · Bu addım nəyə görədir

import { JwtModule } from '@nestjs/jwt';
import { PassportModule } from '@nestjs/passport';
import { AuthService } from './auth.service.js';
import { AuthController } from './auth.controller.js';
import { JwtStrategy } from './strategies/jwt.strategy.js';
import type { SignOptions } from 'jsonwebtoken';

@Module({
  imports: [
    PassportModule,
    /*
     * ⚠️ registerAsync MÜTLƏQDİR — register() YOX.
     *
     * Səbəb: JwtModule.register({...}) modul YÜKLƏNƏNDƏ icra olunur,

        signOptions: {
          expiresIn: (config.get<string>('JWT_MUDDET') ?? '8h') as SignOptions['expiresIn'],
        },
      }),
    }),
  ],
  controllers: [AuthController],
  providers: [AuthService, JwtStrategy],
})

import {
  CallHandler, ExecutionContext, Injectable, Logger, NestInterceptor,
} from '@nestjs/common';
import { Observable, tap } from 'rxjs';
import { PrismaService } from '../../prisma/prisma.service.js';
import type { CariIstifadeci } from '../../auth/decorators/current-user.decorator.js';

/** Yalnız bu HTTP metodları jurnala yazılır */
const IZLENEN_METODLAR = ['POST', 'PATCH', 'PUT', 'DELETE'];

@Injectable()
export class AuditInterceptor implements NestInterceptor {
  private readonly log = new Logger('AUDIT');

  constructor(private readonly prisma: PrismaService) {}

  intercept(context: ExecutionContext, novbeti: CallHandler): Observable<unknown> {
    const sorqu = context.switchToHttp().getRequest();
    const metod: string = sorqu.method;

    // Oxuma əməliyyatlarını yazmırıq — jurnal dolub-daşardı
    if (!IZLENEN_METODLAR.includes(metod)) {
      return novbeti.handle();
    }

    const istifadeci = sorqu.user as CariIstifadeci | undefined;
    const baslama = Date.now();
    const cedvel = this.cedvelAdi(sorqu.route?.path ?? sorqu.url);

    return novbeti.handle().pipe(
      tap({
        next: () =>
          this.yaz(cedvel, metod, istifadeci, sorqu, Date.now() - baslama, null),
        error: (xeta: Error) =>
          this.yaz(cedvel, metod, istifadeci, sorqu, Date.now() - baslama, xeta.message),
      }),
    );
  }

  /**
   * URL-dən cədvəl adını çıxarır.
   *   /api/v1/struktur/merkezler/:id  →  struktur.merkezler
   */
  private cedvelAdi(yol: string): string {
    return (
      yol
        .replace(/^\/api\/v\d+\//, '') // qlobal prefiksi sil
        .replace(/^\//, '')
        .replace(/\/:[^/]+.*$/, '') // :id kimi parametrləri sil
        .replace(/\/+$/, '')
        .replace(/\//g, '.') || 'namelum'
    );
  }

  private async yaz(
    cedvel: string,
    metod: string,
    istifadeci: CariIstifadeci | undefined,
    sorqu: { params?: Record<string, string>; url: string },
    gecikme: number,
    xeta: string | null,
  ): Promise<void> {
    const setirId = sorqu.params?.id ? Number(sorqu.params.id) : null;
    const istifadeciAdi = istifadeci?.email ?? 'anonim';
    const qeyd = xeta ? `XƏTA: ${xeta} | ${gecikme}ms` : `Uğurlu | ${gecikme}ms`;

    try {
      await this.prisma.$executeRaw`
        INSERT INTO audit.audit_log (cedvel_adi, emeliyyat, setir_id, istifadeci, qeyd)
        VALUES (${cedvel}, ${metod}, ${setirId}, ${istifadeciAdi}, ${qeyd})`;
    } catch (x) {
      // ⚠️ Audit yazılmaması ƏSAS əməliyyatı POZMASIN
      this.log.error(`Audit yazılmadı: ${(x as Error).message}`);
    }

    this.log.log(`${metod} ${sorqu.url} — ${istifadeciAdi} — ${qeyd}`);
  }
}

/**
 * İstifadəçiləri yaradır / yeniləyir.
 *
 *   npm run seed:auth
 *
 * İDEMPOTENTDİR: ikinci dəfə işlətsəniz, mövcud istifadəçilərin
 * şifrəsini yeniləyir — dublikat yaratmır.
 */
import 'dotenv/config';
import * as bcrypt from 'bcryptjs';
import { Pool } from 'pg';

interface SeedIstifadeci {
  email: string;
  parol: string;
  ad_soyad: string;
  rol: 'admin' | 'muhendis' | 'maliyyeci' | 'baxici';
}

const ISTIFADECILER: SeedIstifadeci[] = [
  { email: 'admin@arti.edu.az',     parol: '123456', ad_soyad: 'Elnur Əliyev',    rol: 'admin' },
  { email: 'muhendis@arti.edu.az',  parol: '123456', ad_soyad: 'Rəşad Məmmədov',  rol: 'muhendis' },
  { email: 'maliyyeci@arti.edu.az', parol: '123456', ad_soyad: 'Tural İsmayılov', rol: 'maliyyeci' },
  { email: 'baxici@arti.edu.az',    parol: '123456', ad_soyad: 'Günel Rzayeva',   rol: 'baxici' },
];

const BCRYPT_RAUND = 10;

async function main() {
  const pool = new Pool({ connectionString: process.env.DATABASE_URL });

  console.log('════ İstifadəçilər yaradılır ════\n');

  for (const i of ISTIFADECILER) {
    const hash = await bcrypt.hash(i.parol, BCRYPT_RAUND);

    const netice = await pool.query(
      `INSERT INTO kadrlar.istifadeciler (email, parol_hash, ad_soyad, rol, aktiv)
       VALUES (lower($1), $2, $3, $4, true)
       ON CONFLICT (email) DO UPDATE
         SET parol_hash = EXCLUDED.parol_hash,
             ad_soyad   = EXCLUDED.ad_soyad,
             rol        = EXCLUDED.rol,
             aktiv      = true
       RETURNING id, email, rol`,
      [i.email, hash, i.ad_soyad, i.rol],
    );

    const s = netice.rows[0];
    console.log(
      `  ✓ id=${String(s.id).padStart(2)}  ${s.email.padEnd(24)} ` +
        `${s.rol.padEnd(10)} şifrə: ${i.parol}`,
    );
  }

  const cem = await pool.query('SELECT count(*)::int AS say FROM kadrlar.istifadeciler');
  console.log(`\n════ CƏMİ: ${cem.rows[0].say} istifadəçi ════`);

  await pool.end();
}

main().catch((xeta) => {
  console.error('XƏTA:', xeta.message);
  process.exit(1);
});
