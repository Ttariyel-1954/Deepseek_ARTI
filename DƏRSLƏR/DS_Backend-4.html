2 EmbeddingService — mətn → vektor

A · Bu addım nəyə görədir

AI-ın sənədləri «anlaması» üçün onları rəqəmlərə çevirmək lazımdır. EmbeddingService hər mətni 64 ölçülü vektora çevirir; oxşar mətnlərin vektorları bir-birinə yaxın olur. Real layihədə bu işi embedding API-si görür, biz isə FNV-1a hash ilə sadə deterministik versiya yazırıq — beləliklə açar olmadan da RAG-ın necə işlədiyini görürük. Vektor uzunluğu 1-ə normallaşdırılır ki, kosinus sadəcə skalyar hasili olsun.

B · Kod

src/ai/embedding.service.ts

mkdir -p src/ai
cat > src/ai/embedding.service.ts <<'EOF'
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service.js';

/** Vektorun ölçüsü — kiçik saxlanılır ki, JSONB yüngül olsun */
export const OLCU = 64;

export interface EmbeddingSetri {
  id: number;
  cedvel_adi: string;
  sened_id: number | null;
  metn: string | null;
  vektor: number[] | null;
}

/**
 * Mətn → vektor.
 *
 * ⚠️ Bu, SADƏ və DETERMİNİSTİK bir embedding-dir (FNV-1a hash).
 * Real layihədə burada DeepSeek/OpenAI embedding API-si olur. Bizim
 * məqsədimiz RAG-ın NECƏ işlədiyini göstərməkdir — API açarı olmadan da.
 */
@Injectable()
export class EmbeddingService {
  constructor(private readonly prisma: PrismaService) {}

  /** Mətni 64 ölçülü normallaşdırılmış vektora çevirir */
  vektor(metn: string): number[] {
    const v = new Array<number>(OLCU).fill(0);
    const sözler = metn
      .toLowerCase()
      .split(/[^a-zəğıöüçş0-9]+/u)
      .filter((s) => s.length > 1);

    for (const söz of sözler) {
      let h = 2166136261; // FNV-1a başlanğıc dəyəri
      for (let i = 0; i < söz.length; i++) {
        h ^= söz.charCodeAt(i);
        h = Math.imul(h, 16777619); // FNV əsası
      }
      v[Math.abs(h) % OLCU] += 1;
    }

    const uzunluq = Math.sqrt(v.reduce((c, x) => c + x * x, 0)) || 1;
    return v.map((x) => Number((x / uzunluq).toFixed(6)));
  }

  /** İki vektor arasında kosinus oxşarlığı (−1 … 1) */
  kosinus(a: number[], b: number[]): number {
    let cem = 0;
    for (let i = 0; i < Math.min(a.length, b.length); i++) cem += a[i] * b[i];
    return Number(cem.toFixed(6));
  }

  /** Mətnı bazaya vektorlaşdırılmış şəkildə yazır */
  async saxla(cedvelAdi: string, senedId: number | null, metn: string) {
    const vektör = this.vektor(metn);
    const setirler = await this.prisma.$queryRaw<{ id: number }[]>`
      INSERT INTO ai.embeddingler (cedvel_adi, sened_id, metn, vektor)
      VALUES (${cedvelAdi}, ${senedId}, ${metn}, ${JSON.stringify(vektör)}::jsonb)
      RETURNING id::int AS id`;
    return { id: setirler[0]?.id ?? 0, olcu: OLCU };
  }

  /** Bazadakı bütün embedding-lər */
  async hamisi(cedvelAdi?: string): Promise<EmbeddingSetri[]> {
    if (cedvelAdi) {
      return this.prisma.$queryRaw<EmbeddingSetri[]>`
        SELECT id::int AS id, cedvel_adi, sened_id, metn, vektor
          FROM ai.embeddingler WHERE cedvel_adi = ${cedvelAdi} ORDER BY id`;
    }
    return this.prisma.$queryRaw<EmbeddingSetri[]>`
      SELECT id::int AS id, cedvel_adi, sened_id, metn, vektor
        FROM ai.embeddingler ORDER BY id`;
  }

  async say(): Promise<number> {
    const s = await this.prisma.$queryRaw<{ say: number }[]>`
      SELECT count(*)::int AS say FROM ai.embeddingler`;
    return s[0]?.say ?? 0;
  }
}
EOF

3 RagService — sənədlər arasında axtarış

A · Bu addım nəyə görədir

RAG (Retrieval-Augmented Generation) belə işləyir: sual vektora çevrilir, bazadakı bütün sənəd vektorları ilə müqayisə edilir və ən oxşar olanlar seçilir. Onlar AI-a «kontekst» kimi verilir. Bu addımda həm axtarış, həm də sened.senedler cədvəlini vektorlaşdıran köməkçi metod yaranır. Vektorlar ai.embeddingler cədvəlində JSONB kimi saxlanılır.

import { Injectable } from '@nestjs/common';
import { EmbeddingService, EmbeddingSetri } from './embedding.service.js';
import { PrismaService } from '../prisma/prisma.service.js';

export interface RagNeticesi {
  sened_id: number | null;
  metn: string;
  oxsarlıq: number;
}

/**
 * RAG — Retrieval-Augmented Generation.
 *
 * Sualı vektora çevirir, bazadakı vektorlarla müqayisə edir və ƏN OXŞAR
 * sənədləri qaytarır. AI həmin sənədləri kontekst kimi alır.
 *
 * ⚠️ Vektor axtarışı TƏTBİQ TƏRƏFİNDƏ aparılır (kosinus), çünki `pgvector`
 * genişlənməsi superuser tələb edir. 10 000+ sənəddə bu yavaş olar — o zaman
 * pgvector və ya xüsusi vektor bazası lazımdır.
 */
@Injectable()
export class RagService {
  constructor(
    private readonly embedding: EmbeddingService,
    private readonly prisma: PrismaService,
  ) {}

  /** Suala ən oxşar sənədləri tapır */
  async axtar(
    sual: string,
    limit = 3,
    cedvelAdi = 'sened.senedler',
  ): Promise<RagNeticesi[]> {
    const sualVektoru = this.embedding.vektor(sual);
    const setirler = await this.embedding.hamisi(cedvelAdi);

    return setirler
      .filter(
        (s): s is EmbeddingSetri & { vektor: number[]; metn: string } =>
          Array.isArray(s.vektor) && typeof s.metn === 'string',
      )
      .map((s) => ({
        sened_id: s.sened_id,
        metn: s.metn,
        oxsarlıq: this.embedding.kosinus(sualVektoru, s.vektor),
      }))
      .filter((n) => n.oxsarlıq > 0)
      .sort((a, b) => b.oxsarlıq - a.oxsarlıq)
      .slice(0, limit);
  }

  /** `sened.senedler` cədvəlini vektorlaşdırır */
  async vektorlasdir(): Promise<{ baxildi: number; yazildi: number }> {
    const senedler = await this.prisma.$queryRaw<
      { id: number; movzu: string; qeyd: string | null }[]
    >`
      SELECT id::int AS id, movzu, qeyd
        FROM sened.senedler ORDER BY id`;

    let yazildi = 0;
    for (const s of senedler) {
      const metn = `${s.movzu}${s.qeyd ? ' — ' + s.qeyd : ''}`;
      await this.embedding.saxla('sened.senedler', s.id, metn);
      yazildi++;
    }

    return { baxildi: senedler.length, yazildi };
  }
}
EOF

    return null;

        uygun: false,
        izah:

          RESEPTLER.map((x) => x.izah).join('; '),
        setirler: [],
      };

    return { uygun: true, izah: r.izah, setirler: await this.icra(r.izah) };

                 count(DISTINCT s.id)::int AS shobe_sayi,
                 count(DISTINCT e.id)::int AS emekdas_sayi
            FROM struktur.merkezler m
            LEFT JOIN struktur.shobeler  s ON s.merkez_id = m.id
            LEFT JOIN kadrlar.emekdaslar e ON e.merkez_id = m.id
           GROUP BY m.ad ORDER BY m.ad`;

      case 'Ən çox maaş alan 5 nəfər':

           ORDER BY e.maas DESC LIMIT 5`;

      case 'Orta əmək haqqı':
        return this.prisma.$queryRaw`
          SELECT round(avg(maas)::numeric, 2)::float8 AS orta_maas
            FROM kadrlar.emekdaslar WHERE maas IS NOT NULL`;

      case 'Vəzifələr üzrə bölgü':
        return this.prisma.$queryRaw`
          SELECT v.ad, count(*)::int AS say
            FROM kadrlar.emekdaslar e
            JOIN struktur.vezifeler v ON v.id = e.vezife_id
           GROUP BY v.ad ORDER BY count(*) DESC`;

      case 'Elmi dərəcələr üzrə bölgü':
        return this.prisma.$queryRaw`
          SELECT COALESCE(d.ad, '— yoxdur —') AS ad, count(*)::int AS say
            FROM kadrlar.emekdaslar e
            LEFT JOIN ortaq.elmi_dereceler d ON d.id = e.elmi_derece_id
           GROUP BY d.ad ORDER BY count(*) DESC`;

      case 'Şöbələrin sayı':
        return this.prisma.$queryRaw`
          SELECT count(*)::int AS shobe_sayi FROM struktur.shobeler`;

      case 'Sənədlərin növlər üzrə bölgüsü':
        return this.prisma.$queryRaw`
          SELECT n.ad, count(*)::int AS say
            FROM sened.senedler s
            JOIN sened.sened_novleri n ON n.id = s.nov_id
           GROUP BY n.ad ORDER BY count(*) DESC`;

      case 'Aktiv / deaktiv əməkdaşlar':
        return this.prisma.$queryRaw`
          SELECT CASE WHEN aktiv THEN 'Aktiv' ELSE 'Deaktiv' END AS status,
                 count(*)::int AS say
            FROM kadrlar.emekdaslar GROUP BY aktiv ORDER BY aktiv DESC`;

      case 'Müqavilələrin sayı':
        return this.prisma.$queryRaw`
          SELECT count(*)::int AS muqavile_sayi FROM maliyye.muqavileler`;

      default:
        return [];
    }
  }

import { Injectable } from '@nestjs/common';
import ExcelJS from 'exceljs';
import { PrismaService } from '../prisma/prisma.service.js';

/**
 * Excel ixracı.
 *
 * ⚠️ `import ExcelJS from 'exceljs'` — DEFAULT import olmalıdır.
 * `import * as ExcelJS` yazsaq ESM/CJS qarışıqlığı yaranır və
 * `ExcelJS.Workbook is not a constructor` xətası çıxır.

@Injectable()
export class ExcelService {
  constructor(private readonly prisma: PrismaService) {}

  /** Mərkəzlər — Excel faylı Buffer kimi */
  async merkezler(): Promise<Buffer> {
    const setirler = await this.prisma.$queryRaw<
      { id: number; ad: string; tip: string; shobe_sayi: number; emekdas_sayi: number }[]
    >`
      SELECT m.id::int AS id, m.ad, m.tip,
             count(DISTINCT s.id)::int AS shobe_sayi,
             count(DISTINCT e.id)::int AS emekdas_sayi
        FROM struktur.merkezler m
        LEFT JOIN struktur.shobeler  s ON s.merkez_id = m.id
        LEFT JOIN kadrlar.emekdaslar e ON e.merkez_id = m.id
       GROUP BY m.id, m.ad, m.tip ORDER BY m.ad`;

    return this.yarat('Mərkəzlər', [
      { başlıq: 'ID', açar: 'id', en: 8 },
      { başlıq: 'Ad', açar: 'ad', en: 45 },
      { başlıq: 'Tip', açar: 'tip', en: 14 },
      { başlıq: 'Şöbə sayı', açar: 'shobe_sayi', en: 12 },
      { başlıq: 'Əməkdaş sayı', açar: 'emekdas_sayi', en: 14 },
    ], setirler);
  }

  /** Əməkdaşlar — Excel faylı Buffer kimi */
  async emekdaslar(): Promise<Buffer> {
    const setirler = await this.prisma.$queryRaw<
      Record<string, unknown>[]
    >`
      SELECT e.id::int AS id, e.ad, e.soyad, e.ata_adi, e.email,
             e.maas::float8 AS maas, e.aktiv,
             m.ad AS merkez, s.ad AS shobe, v.ad AS vezife
        FROM kadrlar.emekdaslar e
        LEFT JOIN struktur.merkezler m ON m.id = e.merkez_id
        LEFT JOIN struktur.shobeler  s ON s.id = e.shobe_id
        LEFT JOIN struktur.vezifeler v ON v.id = e.vezife_id
       ORDER BY e.soyad, e.ad`;

    return this.yarat('Əməkdaşlar', [
      { başlıq: 'ID', açar: 'id', en: 8 },
      { başlıq: 'Ad', açar: 'ad', en: 16 },
      { başlıq: 'Soyad', açar: 'soyad', en: 18 },
      { başlıq: 'Ata adı', açar: 'ata_adi', en: 16 },
      { başlıq: 'E-poçt', açar: 'email', en: 30 },
      { başlıq: 'Maaş', açar: 'maas', en: 12 },
      { başlıq: 'Mərkəz', açar: 'merkez', en: 34 },
      { başlıq: 'Şöbə', açar: 'shobe', en: 30 },
      { başlıq: 'Vəzifə', açar: 'vezife', en: 24 },
      { başlıq: 'Aktiv', açar: 'aktiv', en: 8 },
    ], setirler);
  }

  /** Ümumi Excel qurucusu */
  private async yarat(
    vərəqAdı: string,
    sütunlar: { başlıq: string; açar: string; en: number }[],
    setirler: Record<string, unknown>[],
  ): Promise<Buffer> {
    const kitab = new ExcelJS.Workbook();
    kitab.creator = 'ARTİ ERP';
    kitab.created = new Date();

    const vərəq = kitab.addWorksheet(vərəqAdı);
    vərəq.columns = sütunlar.map((s) => ({
      header: s.başlıq,
      key: s.açar,
      width: s.en,
    }));

    // Başlıq sətri — qalın və boz fon
    vərəq.getRow(1).font = { bold: true };
    vərəq.getRow(1).fill = {
      type: 'pattern',
      pattern: 'solid',
      fgColor: { argb: 'FFE2E8F0' },
    };

    for (const s of setirler) vərəq.addRow(s);
    vərəq.autoFilter = { from: 'A1', to: { row: 1, column: sütunlar.length } };

    const bufer = await kitab.xlsx.writeBuffer();
    return Buffer.from(bufer);
  }
}

# ── KÖMƏKÇİLƏR ────────────────────────────────────────────────────────
bashliq() {
  printf "\n${MAVI}════════════════════════════════════════════════════════════${SIFIR}\n"
  printf "${MAVI}  %s${SIFIR}\n" "$1"
  printf "${MAVI}════════════════════════════════════════════════════════════${SIFIR}\n"
}

yoxla() {   # yoxla "ad" "gözlənilən" "faktiki"
  if [ "$2" = "$3" ]; then
    printf "  ${YASIL}✓${SIFIR} %-50s ${BOZ}%s${SIFIR}\n" "$1" "$3"
    KECDI=$((KECDI + 1))
  else
    printf "  ${QIRMIZI}✗${SIFIR} %-50s ${QIRMIZI}%s${SIFIR} ${BOZ}(gözlənilən: %s)${SIFIR}\n" \
      "$1" "$3" "$2"
    XETA=$((XETA + 1))
  fi
}

# kod "metod" "yol" [token]
kod() {
  if [ -n "${3:-}" ]; then
    curl -s -o /dev/null -w '%{http_code}' -X "$1" "$A/$2" \
      -H "Authorization: Bearer $3" 2>/dev/null
  else
    curl -s -o /dev/null -w '%{http_code}' -X "$1" "$A/$2" 2>/dev/null
  fi
}

# kod_cisim "yol" "cisim" [token]
kod_cisim() {
  if [ -n "${3:-}" ]; then
    curl -s -o /dev/null -w '%{http_code}' -X POST "$A/$1" \
      -H 'Content-Type: application/json' \
      -H "Authorization: Bearer $3" -d "$2" 2>/dev/null
  else
    curl -s -o /dev/null -w '%{http_code}' -X POST "$A/$1" \
      -H 'Content-Type: application/json' -d "$2" 2>/dev/null
  fi
}

# getir "yol" [token] → gövdə
getir() {
  if [ -n "${2:-}" ]; then
    curl -s "$A/$1" -H "Authorization: Bearer $2" 2>/dev/null
  else
    curl -s "$A/$1" 2>/dev/null
  fi
}

# gonder "yol" "cisim" [token] → gövdə
gonder() {
  if [ -n "${3:-}" ]; then
    curl -s -X POST "$A/$1" -H 'Content-Type: application/json' \
      -H "Authorization: Bearer $3" -d "$2" 2>/dev/null
  else
    curl -s -X POST "$A/$1" -H 'Content-Type: application/json' -d "$2" 2>/dev/null
  fi
}

# sahe "cisim" "açar" → sadə skalyar dəyər
sahe() {
  printf '%s' "$1" | grep -o "\"$2\":[^,}]*" | head -1 | sed "s/\"$2\"://" \
    | sed 's/^"//; s/"$//'
}

# sayi "cisim" "açar" → massivdəki element sayı (sadə üsul)
sayi() {
  printf '%s' "$1" | grep -o "\"$2\"" | wc -l | tr -d ' '
}

psq() { psql -U "$DB_ISTIFADECI" -d "$BAZA" -tA -c "$1" 2>/dev/null; }
